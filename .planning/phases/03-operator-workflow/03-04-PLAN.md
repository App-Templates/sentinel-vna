```typescript
export const useSubmitFeedback = () => useMutation({
  mutationFn: async ({ alertId, feedbackType, comment }: { alertId: string; feedbackType: FeedbackType; comment?: string }) => {
    const res = await fetch(`/api/alerts/${alertId}/feedback`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ feedback_type: feedbackType, comment }),
    });
    if (!res.ok) throw new Error('Failed to submit feedback');
    return res.json();
  },
  onSuccess: () => queryClient.invalidateQueries({ queryKey: ['alert'] }),
});

export const useAssignAlert = () => useMutation({
  mutationFn: async ({ alertId, assignedTo }: { alertId: string; assignedTo: string }) => {
    const res = await fetch(`/api/alerts/${alertId}/assign`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ assigned_to: assignedTo }),
    });
    if (!res.ok) throw new Error('Failed to assign alert');
    return res.json();
  },
  onSuccess: () => queryClient.invalidateQueries({ queryKey: ['alert', 'alerts'] }),
});

export const useUnassignAlert = () => useMutation({
  mutationFn: async (alertId: string) => {
    const res = await fetch(`/api/alerts/${alertId}/assign`, { method: 'DELETE' });
    if (!res.ok) throw new Error('Failed to unassign');
    return res.json();
  },
  onSuccess: () => queryClient.invalidateQueries({ queryKey: ['alert', 'alerts'] }),
});

export const useCreateIntervention = () => useMutation({
  mutationFn: async (data: { timestamp: string; publisher_id: string; intervention_type: InterventionType; description: string; value_before?: string; value_after?: string }) => {
    const res = await fetch('/api/interventions', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(data),
    });
    if (!res.ok) throw new Error('Failed to create intervention');
    return res.json();
  },
});

export const useLinkIntervention = () => useMutation({
  mutationFn: async ({ alertId, interventionId }: { alertId: string; interventionId: number }) => {
    const res = await fetch(`/api/alerts/${alertId}/link-intervention`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ intervention_id: interventionId }),
    });
    if (!res.ok) throw new Error('Failed to link intervention');
    return res.json();
  },
  onSuccess: () => queryClient.invalidateQueries({ queryKey: ['alert'] }),
});

export const usePublisherFeedbackHistory = (publisherId?: string) => useQuery({
  queryKey: ['publisher-feedback', publisherId],
  queryFn: async () => {
    if (!publisherId) return { feedbacks: [], total: 0 };
    const res = await fetch(`/api/publishers/${publisherId}/feedback-history?limit=50`);
    if (!res.ok) throw new Error('Failed to load feedback history');
    return res.json();
  },
  enabled: !!publisherId,
});
```
