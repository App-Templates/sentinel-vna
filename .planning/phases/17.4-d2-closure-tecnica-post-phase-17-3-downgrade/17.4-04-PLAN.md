    #!/usr/bin/env python3
    """
    Phase 17.4 D2CLOSE-03 (D-07, D-09) — In-place rewrite of 39 GOOGLE_ADEX
    backfill alerts. Uses Plan 02 Path A (daily_kpi.csv direct read) via
    `backend.agents.data_fetcher_real.data_fetcher_agent` in BACKTEST mode.

    Replaces the synthetic-era revenue_current/baseline/delta_pct with PBI-parity
    numbers; flips data_source='backfill' → 'real_csv'.

    Idempotent: second run completes 0-UPDATE.
    Dry-run: --dry-run prints per-alert before/after without writing.

    Run:
        .venv/bin/python scripts/rewrite_backfill_google_adex_post_17_4.py [--dry-run] [--db <path>]
    """
    from __future__ import annotations

    import argparse
    import sqlite3
    import sys
    from datetime import datetime, timezone
    from pathlib import Path

    REPO_ROOT = Path(__file__).resolve().parent.parent
    if str(REPO_ROOT) not in sys.path:
        sys.path.insert(0, str(REPO_ROOT))

    DEFAULT_DB = REPO_ROOT / "data" / "vna_alerts.db"
    REPORT_PATH = (
        REPO_ROOT
        / ".planning" / "phases"
        / "17.4-d2-closure-tecnica-post-phase-17-3-downgrade"
        / "17.4-BACKFILL-REWRITE-REPORT.md"
    )

    SELECT_TARGETS_SQL = """
        SELECT alert_id, timestamp, channel_group, alert_rule, severity,
               revenue_current, revenue_baseline, revenue_delta_pct, data_source
          FROM alerts
         WHERE channel_group = 'GOOGLE_ADEX'
           AND data_source = 'backfill'
           AND deprecated = 0
         ORDER BY timestamp
    """

    UPDATE_SQL = """
        UPDATE alerts
           SET revenue_current = ?,
               revenue_baseline = ?,
               revenue_delta_pct = ?,
               data_source = 'real_csv',
               updated_at = ?
         WHERE alert_id = ?
    """

    EXPECTED_COUNT = 39


    def _now_canonical() -> str:
        """Migration 004 canonical: space-separator naive UTC microseconds."""
        return datetime.now(timezone.utc).replace(tzinfo=None).strftime(
            "%Y-%m-%d %H:%M:%S.%f"
        )


    def _build_state(row: sqlite3.Row) -> dict:
        # Convert DB-format timestamp ('YYYY-MM-DD HH:MM:SS' or similar) to ISO UTC.
        ts = row["timestamp"]
        if "T" not in ts and " " in ts:
            ts = ts.replace(" ", "T")
        if "+" not in ts and "Z" not in ts and "." in ts:
            ts = ts + "+00:00" if ts.count(":") >= 2 else ts
        elif "+" not in ts and "Z" not in ts:
            ts = ts + "+00:00"
        return {
            "trigger_type": row["alert_rule"],
            "channel_group": "GOOGLE_ADEX",
            "mode": "BACKTEST",
            "timestamp": ts,
            "execution_errors": [],
        }


    def _compute_new_metrics(row: sqlite3.Row) -> tuple[float, float, float]:
        from backend.agents.data_fetcher_real import data_fetcher_agent
        state = _build_state(row)
        result = data_fetcher_agent(state)
        cur = (result.get("current_metrics") or {}).get("revenue", 0.0)
        base = (result.get("baseline_metrics") or {}).get("revenue", 0.0)
        delta_pct = ((cur - base) / base * 100.0) if base > 0 else 0.0
        # Sanity check: Plan 02 must have set data_source='real_csv'
        ds = result.get("data_source", "?")
        if ds != "real_csv":
            raise RuntimeError(
                f"alert {row['alert_id']}: Plan 02 Path A did not activate "
                f"(data_source={ds!r}, expected 'real_csv'). Abort to avoid bad UPDATE."
            )
        return round(float(cur), 2), round(float(base), 2), round(float(delta_pct), 2)


    def _emit_report(rows: list[dict], db_path: Path) -> None:
        lines = [
            "---",
            "phase: 17.4-d2-closure-tecnica-post-phase-17-3-downgrade",
            "type: backfill-rewrite-report",
            f"generated: {_now_canonical()}Z",
            f"db_path: {db_path}",
            f"alerts_rewritten: {len(rows)}",
            "channel_group: GOOGLE_ADEX",
            "rewrite_source: data/sentinel_db/daily_kpi.csv (via Plan 02 Path A)",
            "---",
            "",
            "# Phase 17.4 — Backfill Rewrite Report (D2CLOSE-03)",
            "",
            f"**Generated:** {_now_canonical()} UTC",
            f"**Total alerts UPDATEd:** {len(rows)}",
            "**Source of new values:** `data/sentinel_db/daily_kpi.csv` aggregated by (date, channel_group='GOOGLE_ADEX') via Plan 02 `_compute_network_totals_for_channel`.",
            "**Provenance flip:** `data_source='backfill'` → `data_source='real_csv'`.",
            "",
            "## Per-alert before/after",
            "",
            "| alert_id | timestamp | old_revenue (€) | new_revenue (€) | delta_eur (€) | delta_pct (%) |",
            "|----------|-----------|-----------------|-----------------|---------------|---------------|",
        ]
        total_delta_eur = 0.0
        max_delta_eur = 0.0
        for r in rows:
            d_eur = r["new_revenue"] - r["old_revenue"]
            d_pct = (d_eur / r["old_revenue"] * 100.0) if r["old_revenue"] != 0 else 0.0
            total_delta_eur += d_eur
            max_delta_eur = max(max_delta_eur, abs(d_eur))
            lines.append(
                f"| {r['alert_id']} | {r['timestamp']} | {r['old_revenue']:.2f} | "
                f"{r['new_revenue']:.2f} | {d_eur:+.2f} | {d_pct:+.2f}% |"
            )
        lines.extend([
            "",
            "## Summary statistics",
            "",
            f"- Alerts rewritten: {len(rows)}",
            f"- Mean delta EUR: {(total_delta_eur / len(rows)) if rows else 0.0:+.2f}",
            f"- Max abs delta EUR: {max_delta_eur:.2f}",
            f"- Source CSV: `data/sentinel_db/daily_kpi.csv` (no top-50 filter; Phase 11 SUMMARY verified PBI parity ±0.17%)",
            f"- Helper: `backend.api.routes._compute_network_totals_for_channel(date, 'GOOGLE_ADEX', bypass_top50=True)`",
            f"- Phase 17.3 root cause (tz-mismatch silent synthetic fallback) closed in Plan 02; this rewrite makes the closure visible in historical alert rows.",
            "",
            "## Rollback",
            "",
            "1. Selective rollback: `sqlite3 data/vna_alerts.db < scripts/migrations/006_rollback.sql` (this only undoes Migration 006 marking, NOT this rewrite)",
            "2. Full rollback: `cp data/vna_alerts.db.pre-17.4-backfill-rewrite data/vna_alerts.db`",
            "3. Code rollback: `git checkout pre-17.4-backfill-rewrite` (reverts Migration 006 + driver source; live DB only restored via #2)",
        ])
        REPORT_PATH.write_text("\n".join(lines) + "\n")

        ap.add_argument("--dry-run", action="store_true")
        ap.add_argument("--db", type=Path, default=DEFAULT_DB)
        args = ap.parse_args()

        if not args.db.exists():
            print(f"FATAL: DB not found at {args.db}")
            return 1

        conn = sqlite3.connect(str(args.db))
        conn.row_factory = sqlite3.Row
        try:
            rows = conn.execute(SELECT_TARGETS_SQL).fetchall()
        finally:
            pass

        if len(rows) == 0:
            # Could be already-rewritten or 0 targets — check the post-state.
            post = conn.execute(
                "SELECT COUNT(*) FROM alerts WHERE channel_group='GOOGLE_ADEX' AND data_source='real_csv'"
            ).fetchone()[0]
            conn.close()
            if post == EXPECTED_COUNT:
                print(f"IDEMPOTENT-NOOP: all {EXPECTED_COUNT} GOOGLE_ADEX backfill alerts already rewritten.")
                return 0
            print(f"WARNING: 0 targets and {post} already-rewritten (expected {EXPECTED_COUNT}). Manual inspection required.")
            return 1

        if len(rows) != EXPECTED_COUNT:
            print(f"WARNING: expected {EXPECTED_COUNT} targets, found {len(rows)}. "
                  f"Proceeding but report may be partial.")

        records = []
        for i, row in enumerate(rows, 1):
            try:
                new_cur, new_base, new_delta_pct = _compute_new_metrics(row)
            except Exception as e:
                print(f"[{i}/{len(rows)}] {row['alert_id']} FAILED: {e}")
                conn.close()
                return 2

            old_cur = row["revenue_current"]
            print(
                f"[{i}/{len(rows)}] {row['alert_id']} {row['timestamp']} "
                f"old=€{old_cur:.2f} new=€{new_cur:.2f} delta=€{new_cur - old_cur:+.2f}"
            )
            records.append({
                "alert_id": row["alert_id"],
                "timestamp": row["timestamp"],
                "old_revenue": old_cur,
                "new_revenue": new_cur,
                "new_baseline": new_base,
                "new_delta_pct": new_delta_pct,
            })

            if not args.dry_run:
                conn.execute(
                    UPDATE_SQL,
                    (new_cur, new_base, new_delta_pct, _now_canonical(), row["alert_id"]),
                )

        if not args.dry_run:
            conn.commit()

        # Post-write verification
        post_count = conn.execute(
            "SELECT COUNT(*) FROM alerts WHERE channel_group='GOOGLE_ADEX' AND data_source='real_csv'"
        ).fetchone()[0]
        conn.close()

        if args.dry_run:
            print(f"\nDRY RUN: {len(records)} alerts WOULD be UPDATEd. No DB writes.")
            return 0

        if post_count != EXPECTED_COUNT:
            print(f"FAIL: post-rewrite count expected {EXPECTED_COUNT}, got {post_count}")
            return 3

        _emit_report(records, args.db)
        print(f"\nOK: rewrote {len(records)} alerts. Report: {REPORT_PATH}")
        return 0


    if __name__ == "__main__":
        sys.exit(main())
