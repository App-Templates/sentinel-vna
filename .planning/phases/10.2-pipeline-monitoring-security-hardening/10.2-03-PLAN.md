from these would inherit the bug. Fix once, fix everywhere.

def _restore(bak: Path, dst: Path) -> None:
    """Restore the live DB atomically: backup to temp path, then atomic-rename.
    The live DB path is always either the pre-failure state (if killed before
    os.replace) or the fully restored state (if killed after) — never absent
    or partial. MONITOR-14 / WR-03 (Phase 10.2 Plan 03).

    for suffix in ("-wal", "-shm"):
        stale = dst.with_name(dst.name + suffix)
        if stale.exists():
            stale.unlink()

    tmp = dst.with_suffix(dst.suffix + ".restore.tmp")
    if tmp.exists():
        tmp.unlink()

    bak_conn = sqlite3.connect(str(bak))
    dst_conn = sqlite3.connect(str(tmp))
    try:
        bak_conn.backup(dst_conn)
    finally:
        dst_conn.close()
        bak_conn.close()

    os.replace(tmp, dst)
    _log(f"Restored {dst} from {bak} via sqlite3.backup() + atomic replace")

    class TestMigration004Restore(unittest.TestCase):
        """MONITOR-14 / WR-03 (Phase 10.2 Plan 03): atomic _restore. A kill (SIGKILL,
        OOM, power-loss) between dst.unlink() and bak_conn.backup() completion must
        not leave the live DB absent or empty. The fix uses a temp path + os.replace
        for atomicity; this regression test simulates the kill via mock.patch.
        """

        def setUp(self):
            import importlib.util
            self.tmpdir = Path(tempfile.mkdtemp(prefix="vna_test_restore_"))
            self.dst = self.tmpdir / "live.db"
            self.bak = self.tmpdir / "backup.db"

            # Create a minimal valid SQLite DB at dst (the "live" DB).
            conn = sqlite3.connect(str(self.dst))
            conn.executescript(
                "CREATE TABLE marker(id INTEGER PRIMARY KEY, label TEXT);"
                "INSERT INTO marker(label) VALUES ('original-live-db');"
            )
            conn.commit()
            conn.close()

            self.pre_sha = hashlib.sha256(self.dst.read_bytes()).hexdigest()

            # Build a separate "backup" file via sqlite3.Connection.backup so bak is
            # a real, valid SQLite DB (mirrors the production backup() pattern).
            src = sqlite3.connect(str(self.dst))
            tgt = sqlite3.connect(str(self.bak))
            try:
                src.backup(tgt)
            finally:
                tgt.close()
                src.close()

            # Load the apply_migration_004 driver module by file path.
            spec = importlib.util.spec_from_file_location(
                "apply_migration_004_for_restore_test",
                REPO_ROOT / "scripts" / "apply_migration_004.py",
            )
            self.mod = importlib.util.module_from_spec(spec)
            spec.loader.exec_module(self.mod)

        def tearDown(self):
            shutil.rmtree(self.tmpdir, ignore_errors=True)

        def test_restore_atomic_intermediate_failure_leaves_db_intact(self):
            """Simulate a kill between sqlite3.connect(tmp) and bak_conn.backup()
            completing. Post-fix: dst is unchanged (os.replace never ran). Pre-fix
            (no atomic rename): dst would be absent or empty."""
            with mock.patch.object(
                sqlite3.Connection, "backup",
                side_effect=RuntimeError("simulated kill mid-restore"),
            ):
                with self.assertRaises(RuntimeError):
                    self.mod._restore(self.bak, self.dst)

            self.assertTrue(
                self.dst.exists(),
                "WR-03 regression: live DB absent post-kill (pre-fix would unlink)",
            )
            post_sha = hashlib.sha256(self.dst.read_bytes()).hexdigest()
            self.assertEqual(
                self.pre_sha, post_sha,
                "WR-03 regression: live DB modified post-kill (atomicity violated)",
            )

        def test_restore_success_replaces_dst_with_backup_contents(self):
            """Sanity: a non-failing _restore call DOES update dst with bak contents.
            Mutate dst post-bak so dst != bak; call _restore; assert dst content
            matches the pre-mutation state captured in bak."""
            conn = sqlite3.connect(str(self.dst))
            conn.execute("INSERT INTO marker(label) VALUES ('post-backup-row')")
            conn.commit()
            conn.close()
            mutated_sha = hashlib.sha256(self.dst.read_bytes()).hexdigest()
            self.assertNotEqual(self.pre_sha, mutated_sha)

            conn = sqlite3.connect(str(self.dst))

            conn.close()
            self.assertEqual(
                rows, [("original-live-db",)],
                "WR-03 fix: _restore should fully replace dst with bak contents",
            )
