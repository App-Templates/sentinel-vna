Create a new component that displays drift regression results with a visual trend line:

```typescript
import { Card } from '../ui/Card';
import { DeltaIndicator } from '../ui/DeltaIndicator';
import type { DriftRegression } from '../../types/alert';
import { LineChart, Line, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer } from 'recharts';

interface Props {
  drift: DriftRegression;
}

export function DriftRegressionSection({ drift }: Props) {
  // Generate synthetic trend data points for visualization
  // Backend provides slope and days, we reconstruct the line
  const trendData = Array.from({ length: drift.days_analyzed }, (_, i) => {
    // Approximate revenue trend: assume 100 as base, apply slope
    const dayValue = 100 + (drift.slope_pct_per_day * i);
    return {
      day: i + 1,
      revenue: Math.max(0, dayValue),
    };
  });

  // Add projection point 7 days out
  const projectionPoint = {
    day: drift.days_analyzed + 7,
    revenue: Math.max(0, 100 + (drift.slope_pct_per_day * (drift.days_analyzed + 7))),
    projected: true,
  };
  trendData.push(projectionPoint);

  return (
    <Card className="p-6">
      <h3 className="text-lg font-semibold mb-4">Drift Regression Analysis</h3>

      <div className="grid grid-cols-2 md:grid-cols-4 gap-4 mb-6">
        <div>
          <div className="text-sm text-gray-500 dark:text-gray-400">Slope (per day)</div>
          <div className="text-xl font-semibold flex items-center gap-2">
            <DeltaIndicator value={drift.slope_pct_per_day} />
            <span>{drift.slope_pct_per_day.toFixed(2)}%</span>
          </div>
        </div>

        <div>
          <div className="text-sm text-gray-500 dark:text-gray-400">Cumulative Drop</div>
          <div className="text-xl font-semibold flex items-center gap-2">
            <DeltaIndicator value={drift.cumulative_drop_pct} />
            <span>{drift.cumulative_drop_pct.toFixed(1)}%</span>
          </div>
        </div>

        <div>
          <div className="text-sm text-gray-500 dark:text-gray-400">R² (fit quality)</div>
          <div className="text-xl font-semibold">{drift.r_squared.toFixed(3)}</div>
          <div className="text-xs text-gray-500">
            {drift.r_squared >= 0.8 ? 'Excellent' : drift.r_squared >= 0.5 ? 'Good' : 'Moderate'}
          </div>
        </div>

        <div>
          <div className="text-sm text-gray-500 dark:text-gray-400">7d Projection</div>
          <div className="text-xl font-semibold flex items-center gap-2">
            <DeltaIndicator value={drift.projection_7d_pct} />
            <span>{drift.projection_7d_pct.toFixed(1)}%</span>
          </div>
        </div>
      </div>

      <div className="mt-6">
        <h4 className="text-sm font-medium mb-3">Trend over {drift.days_analyzed} days</h4>
        <ResponsiveContainer width="100%" height={250}>
          <LineChart data={trendData}>
            <CartesianGrid strokeDasharray="3 3" />
            <XAxis
              dataKey="day"
              label={{ value: 'Days analyzed', position: 'insideBottom', offset: -5 }}
            />
            <YAxis
              label={{ value: 'Revenue (indexed)', angle: -90, position: 'insideLeft' }}
            />
            <Tooltip
              formatter={(value: number, name: string, props: any) => {
                if (props.payload.projected) {
                  return [`${value.toFixed(1)} (projected)`, 'Revenue'];
                }
                return [value.toFixed(1), 'Revenue'];
              }}
            />
            <Line
              type="monotone"
              dataKey="revenue"
              stroke="#ef4444"
              strokeWidth={2}
              dot={(props: any) => {
                if (props.payload.projected) {
                  return <circle cx={props.cx} cy={props.cy} r={4} fill="#fbbf24" stroke="#ef4444" strokeWidth={2} />;
                }
                return <circle cx={props.cx} cy={props.cy} r={3} fill="#ef4444" />;
              }}
            />
          </LineChart>
        </ResponsiveContainer>
        <p className="text-xs text-gray-500 mt-2">
          ⚠️ Yellow point indicates 7-day projection based on current trend
        </p>
      </div>

      <div className="mt-4 p-4 bg-amber-50 dark:bg-amber-900/20 rounded-md">
        <p className="text-sm text-amber-800 dark:text-amber-200">
          <strong>Gradual decline detected:</strong> Revenue has been dropping at an average rate of{' '}
          {Math.abs(drift.slope_pct_per_day).toFixed(2)}% per day over the past {drift.days_analyzed} days.
          If this trend continues, expect an additional {Math.abs(drift.projection_7d_pct).toFixed(1)}% drop over the next 7 days.
        </p>
      </div>
    </Card>
  );
}
```

This component:
1. Shows key metrics in a 4-column grid (slope, cumulative, R², projection)
2. Renders a Recharts line chart showing the downward trend
3. Highlights the 7-day projection point in yellow
4. Provides interpretive text for ops team

```typescript
import { Card } from '../ui/Card';
import { Badge } from '../ui/Badge';
import { DeltaIndicator } from '../ui/DeltaIndicator';
import type { YesterdayComparison, MovingAverage14d } from '../../types/alert';
import { ArrowRight } from 'lucide-react';

interface Props {
  yesterdayComparison?: YesterdayComparison;
  movingAverage14d?: MovingAverage14d;
}

export function ComparativeMetricsSection({ yesterdayComparison, movingAverage14d }: Props) {
  if (!yesterdayComparison && !movingAverage14d) {
    return null;
  }

  return (
    <div className="space-y-4">
      {yesterdayComparison && (
        <Card className="p-6">
          <div className="flex items-start justify-between mb-4">
            <h3 className="text-lg font-semibold">Yesterday Comparison</h3>
            {yesterdayComparison.triggered && (
              <Badge variant={yesterdayComparison.severity === 'P0' ? 'destructive' : 'warning'}>
                {yesterdayComparison.severity}
              </Badge>
            )}
          </div>

          <div className="flex items-center gap-6">
            <div>
              <div className="text-sm text-gray-500 dark:text-gray-400">Yesterday (same hour)</div>
              <div className="text-2xl font-semibold">
                €{yesterdayComparison.same_hours_revenue_yesterday.toLocaleString()}
              </div>
            </div>

            <ArrowRight className="text-gray-400" size={24} />

            <div>
              <div className="text-sm text-gray-500 dark:text-gray-400">Today (current hour)</div>
              <div className="text-2xl font-semibold">
                €{yesterdayComparison.current_hour_revenue.toLocaleString()}
              </div>
            </div>

            <div className="ml-auto">
              <div className="text-sm text-gray-500 dark:text-gray-400">Change</div>
              <div className="text-2xl font-semibold flex items-center gap-2">
                <DeltaIndicator value={yesterdayComparison.delta_pct} />
                <span>{yesterdayComparison.delta_pct > 0 ? '+' : ''}{yesterdayComparison.delta_pct.toFixed(1)}%</span>
              </div>
            </div>
          </div>

          {yesterdayComparison.triggered && (
            <div className="mt-4 p-3 bg-red-50 dark:bg-red-900/20 rounded-md">
              <p className="text-sm text-red-800 dark:text-red-200">
                ⚠️ Revenue is down {Math.abs(yesterdayComparison.delta_pct).toFixed(1)}% compared to yesterday's same hour.
                This exceeds the operational threshold of -30%.
              </p>
            </div>
          )}
        </Card>
      )}

      {movingAverage14d && (
        <Card className="p-6">
          <div className="flex items-start justify-between mb-4">
            <h3 className="text-lg font-semibold">14-Day Moving Average</h3>
            {movingAverage14d.triggered && (
              <Badge variant="warning">{movingAverage14d.severity}</Badge>
            )}
          </div>

          <div className="flex items-center gap-6">
            <div>
              <div className="text-sm text-gray-500 dark:text-gray-400">14-day average (same hour)</div>
              <div className="text-2xl font-semibold">
                €{movingAverage14d.avg_revenue_14d.toLocaleString()}
              </div>
            </div>

            <ArrowRight className="text-gray-400" size={24} />

            <div>
              <div className="text-sm text-gray-500 dark:text-gray-400">Current hour</div>
              <div className="text-2xl font-semibold">
                €{movingAverage14d.current_revenue.toLocaleString()}
              </div>
            </div>

            <div className="ml-auto">
              <div className="text-sm text-gray-500 dark:text-gray-400">Divergence</div>
              <div className="text-2xl font-semibold flex items-center gap-2">
                <DeltaIndicator value={movingAverage14d.delta_pct} />
                <span>{movingAverage14d.delta_pct > 0 ? '+' : ''}{movingAverage14d.delta_pct.toFixed(1)}%</span>
              </div>
            </div>
          </div>

          {movingAverage14d.divergence_hours > 0 && (
            <div className="mt-4 flex items-center gap-2">
              <Badge variant="outline">
                Sustained for {movingAverage14d.divergence_hours} consecutive hour{movingAverage14d.divergence_hours > 1 ? 's' : ''}
              </Badge>
            </div>
          )}

          {movingAverage14d.triggered && (
            <div className="mt-4 p-3 bg-amber-50 dark:bg-amber-900/20 rounded-md">
              <p className="text-sm text-amber-800 dark:text-amber-200">
                ⚠️ Revenue has diverged {Math.abs(movingAverage14d.delta_pct).toFixed(1)}% below the 14-day average
                for {movingAverage14d.divergence_hours} consecutive hours.
              </p>
            </div>
          )}
        </Card>
      )}
    </div>
  );
}
```
