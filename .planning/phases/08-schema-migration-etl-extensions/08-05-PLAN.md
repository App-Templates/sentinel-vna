#!/usr/bin/env python3
"""
Phase 08 SCHEMA-03 — Mondadori Feb 2026 fill-rate PBI regression.

Asserts `measurable_impressions / bacino` for publisher 'Mondadori' in
February 2026 matches PBI-declared 35.2% within ±1.5 percentage points.

Memory rule (R-13): uses pyarrow with column projection and pushdown
filtering; only narrow filtered slice is materialized to pandas.

Run:  .venv/bin/python scripts/test_mondadori_fill_rate_regression.py
Exit: 0 = within tolerance, 1 = drift detected or test failure.
"""
import hashlib
import sys
import unittest
from datetime import datetime
from pathlib import Path

import pyarrow.parquet as pq
import pyarrow.compute as pc
import pyarrow as pa

REPO_ROOT = Path(__file__).resolve().parent.parent
DAILY = REPO_ROOT / "data" / "baseline" / "daily_aggregated.parquet"
REPORT = REPO_ROOT / ".planning" / "phases" / "08-schema-migration-etl-extensions" / "fill-rate-report.md"

PBI_FILL_RATE = 0.352
PBI_FILL_RATE_TOLERANCE_PPT = 0.015  # 1.5 percentage points absolute
PBI_BACINO_FEB = 133_000_000
PBI_BACINO_TOLERANCE = 0.015  # 1.5% relative

PUBLISHER = "Mondadori"
FEB_2026_PREFIX = "2026-02"


def _sha256(path: Path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1024 * 1024), b""):
            h.update(chunk)
    return h.hexdigest()


def _load_mondadori_feb() -> pa.Table:
    """Load only Mondadori Feb 2026 rows with the needed columns. Memory-safe."""
    # Read just the 7 columns we need
    table = pq.read_table(
        str(DAILY),
        columns=["date", "publisher_id", "bacino", "measurable_impressions",
                 "environment", "geo_cluster", "channel_group"],
    )
    # Filter: publisher_id == 'Mondadori' (EXACT — not Mondadori_npa)
    mask = pc.and_(
        pc.equal(table.column("publisher_id"), PUBLISHER),
        pc.starts_with(table.column("date"), FEB_2026_PREFIX),
    )
    return table.filter(mask)


class TestMondadoriFillRate(unittest.TestCase):
    @classmethod
    def setUpClass(cls):
        if not DAILY.exists():
            raise unittest.SkipTest(f"Parquet missing: {DAILY}")
        cls.t = _load_mondadori_feb()
        cls.row_count = cls.t.num_rows
        # To pandas on the narrow filtered slice — at most ~50k rows × 7 cols
        cls.df = cls.t.to_pandas()
        # Dedup bacino per (date, environment, geo_cluster) because bacino is
        # channel-agnostic upstream but replicated per channel in the join.
        cls.bacino_dedup_sum = (
            cls.df.drop_duplicates(subset=["date", "environment", "geo_cluster"])
                  ["bacino"].sum()
        )
        cls.measurable_sum = cls.df["measurable_impressions"].sum()
        cls.fill_rate = (
            cls.measurable_sum / cls.bacino_dedup_sum
            if cls.bacino_dedup_sum > 0 else float("nan")
        )

    def test_parquet_file_exists(self):
        self.assertTrue(DAILY.exists())

    def test_row_count_is_sane(self):
        # Mondadori Feb 2026 should have 1k+ rows (many channels × many days × envs × geos)
        self.assertGreater(self.row_count, 1000,
                           f"Only {self.row_count} rows for Mondadori Feb 2026 — "
                           f"filter or publisher_id match failed?")

    def test_mondadori_feb_2026_bacino_matches_pbi_133m(self):
        # PBI 133M ± 1.5% relative → [130.975M, 135.025M]
        low = PBI_BACINO_FEB * (1 - PBI_BACINO_TOLERANCE)
        high = PBI_BACINO_FEB * (1 + PBI_BACINO_TOLERANCE)
        self.assertGreaterEqual(self.bacino_dedup_sum, low,
                                f"bacino dedup {self.bacino_dedup_sum:,} < {low:,} (PBI 133M -1.5%)")
        self.assertLessEqual(self.bacino_dedup_sum, high,
                             f"bacino dedup {self.bacino_dedup_sum:,} > {high:,} (PBI 133M +1.5%)")

    def test_measurable_impressions_non_zero(self):
        self.assertGreater(self.measurable_sum, 1_000_000,
                           f"measurable sum {self.measurable_sum:,} implausibly low — SCHEMA-03 regression?")

    def test_mondadori_feb_2026_fill_rate_matches_pbi(self):
        low = PBI_FILL_RATE - PBI_FILL_RATE_TOLERANCE_PPT
        high = PBI_FILL_RATE + PBI_FILL_RATE_TOLERANCE_PPT
        self.assertGreaterEqual(self.fill_rate, low,
                                f"fill_rate {self.fill_rate:.4f} < {low:.4f} (PBI 35.2% -1.5ppt)")
        self.assertLessEqual(self.fill_rate, high,
                             f"fill_rate {self.fill_rate:.4f} > {high:.4f} (PBI 35.2% +1.5ppt)")

    def test_mondadori_npa_excluded(self):
        # Re-load raw to confirm Mondadori_npa exists as a separate publisher_id
        t_all = pq.read_table(str(DAILY), columns=["publisher_id"])
        uniques = set(pc.unique(t_all.column("publisher_id")).to_pylist())
        # Guard: if intel is correct, both exist as separate publisher_ids
        if "Mondadori_npa" in uniques:
            # Confirm our filter excluded them
            npa_in_ours = (self.df["publisher_id"] == "Mondadori_npa").sum()
            self.assertEqual(npa_in_ours, 0, "Filter should exclude Mondadori_npa")

    def test_writes_fill_rate_report(self):
        # Write provenance + result report. Final test so it runs only after
        # all assertions above pass (unittest runs alphabetically by default,
        # 'test_writes_fill_rate_report' sorts after test_mondadori*; this is
        # not strictly a gate but acceptable for our purpose).
        content = f"""# Phase 08 SCHEMA-03 — Mondadori Feb 2026 Fill Rate Regression

**Generated:** {datetime.now().isoformat()}

## Provenance

- Parquet: `data/baseline/daily_aggregated.parquet`
- SHA256: `{_sha256(DAILY)}`
- Parquet mtime: {datetime.fromtimestamp(DAILY.stat().st_mtime).isoformat()}
- Rows used (publisher=Mondadori, Feb 2026): {self.row_count:,}

## Result

| Metric | Value | PBI Reference | Tolerance | Status |
|--------|-------|---------------|-----------|--------|
| bacino (deduplicated sum) | {self.bacino_dedup_sum:,.0f} | 133,000,000 | ±1.5% relative | {'PASS' if PBI_BACINO_FEB*(1-PBI_BACINO_TOLERANCE) <= self.bacino_dedup_sum <= PBI_BACINO_FEB*(1+PBI_BACINO_TOLERANCE) else 'FAIL'} |
| measurable_impressions (sum) | {self.measurable_sum:,.0f} | (internal) | > 1M | {'PASS' if self.measurable_sum > 1_000_000 else 'FAIL'} |
| fill rate (measurable/bacino) | {self.fill_rate*100:.2f}% | 35.20% | ±1.5 ppt absolute | {'PASS' if PBI_FILL_RATE-PBI_FILL_RATE_TOLERANCE_PPT <= self.fill_rate <= PBI_FILL_RATE+PBI_FILL_RATE_TOLERANCE_PPT else 'FAIL'} |

## Interpretation

- Absolute deviation from PBI 35.2%: {(self.fill_rate - PBI_FILL_RATE)*100:+.2f} percentage points
- SCHEMA-03 (preserve measurable_impressions) is the enabler for this metric.
  Before SCHEMA-03, measurable_impressions was dropped at `scripts/etl_csv_to_parquet.py:187/257`
  and this fill rate could not be computed from parquet.

## Dedup Logic

`bacino` is channel-agnostic in the source CSV `daily_adrequests.csv` but gets
replicated per channel in the merge with `daily_kpi.csv` inside the ETL. To
count it once per (date, environment, geo_cluster) tuple:

```python
bacino_dedup = df.drop_duplicates(
    subset=["date", "environment", "geo_cluster"]
)["bacino"].sum()
```

## Future

This test is the prototype for Phase 18 HARD-04 (VALID contract tests — PBI
match validation for all known metrics). Extending pattern:
- Jan 12 2026 GOOGLE_ADEX gross → €27.2k ± 2%
- Network total gross → PBI total ± 2%
- Revenue gross total Feb 2026 → PBI €192k ± 2.5%
"""
        REPORT.parent.mkdir(parents=True, exist_ok=True)
        REPORT.write_text(content, encoding="utf-8")


def main():
    loader = unittest.TestLoader()
    suite = loader.loadTestsFromTestCase(TestMondadoriFillRate)
    runner = unittest.TextTestRunner(verbosity=2)
    result = runner.run(suite)
    print(f"\nParquet: {DAILY}")
    if DAILY.exists():
        print(f"SHA256:  {_sha256(DAILY)}")
    return 0 if result.wasSuccessful() else 1


if __name__ == "__main__":
    sys.exit(main())
