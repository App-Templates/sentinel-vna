```tsx
import { Component, type ErrorInfo, type ReactNode } from 'react';
import { CHART_ERROR_COPY } from '../../lib/i18n-keys';

interface ChartErrorBoundaryProps {
  children: ReactNode;
  height?: number;
  fallbackMessage?: string;
}

interface ChartErrorBoundaryState {
  hasError: boolean;
  error: Error | null;
}

/**
 * Phase 15 — UI Resilience (CORR-10).
 * Per-chart-area class-component ErrorBoundary. Sized to match parent chart
 * container via `style={{ width: '100%', height }}`. Fallback is chart-scoped,
 * NOT full-page (D-15.06: distinct from global ErrorBoundary which uses
 * min-h-screen for App-level catastrophic crashes).
 *
 * Anti-pattern guards (D-15.04):
 *   - NO `window.location.reload()` button — auto-reload masks repeat failures.
 *   - NO try/catch inside chart render functions (a sibling anti-pattern, see
 *     UI-SPEC AC #14). The boundary catches throws at the React tree boundary;
 *     internal try/catch would suppress the error from this boundary.
 *
 * Diagnostic logging: componentDidCatch logs to console.error with the
 * `[ChartErrorBoundary]` prefix so devtools surfaces the failure (logging is
 * additive — error is still caught and fallback renders).
 *
 * Usage (parent composition layer, D-15.02):
 *   <ChartErrorBoundary height={280}>
 *     <RevenueTrendChart data={trendData} />
 *   </ChartErrorBoundary>
 */
export class ChartErrorBoundary extends Component<
  ChartErrorBoundaryProps,
  ChartErrorBoundaryState
> {
  constructor(props: ChartErrorBoundaryProps) {
    super(props);
    this.state = { hasError: false, error: null };
  }

  static getDerivedStateFromError(error: Error): ChartErrorBoundaryState {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, info: ErrorInfo): void {
    console.error('[ChartErrorBoundary]', error, info);
  }

  render(): ReactNode {
    if (this.state.hasError) {
      const height = this.props.height ?? 280;
      const message = this.props.fallbackMessage ?? CHART_ERROR_COPY.generic;
      return (
        <div
          style={{ width: '100%', height }}
          className="flex items-center justify-center rounded-md border border-gray-200 bg-gray-50"
        >
          <p className="text-sm text-gray-500">{message}</p>
        </div>
      );
    }
    return this.props.children;
  }
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| chart render → boundary catch | A chart component throws on malformed data; the boundary state transitions to hasError. The thrown `error.message` is the only datum that crosses from chart-render-tree to boundary-state. |
| boundary state → user-visible UI | Only `CHART_ERROR_COPY.generic` (static i18n constant) or trusted `fallbackMessage` prop is rendered to the user. `error.message` from chart throws is NEVER rendered to user — it goes to `console.error` only. |
| componentDidCatch → console | Diagnostic info (error + componentStack) is sent to `console.error`. This is a developer surface, not a user-visible surface. |

## STRIDE Threat Register

| Threat ID | Category | Component | Severity | Disposition | Mitigation Plan |
|-----------|----------|-----------|----------|-------------|-----------------|
| T-15-02-01 | Tampering | ChartErrorBoundary fallback render | medium | mitigate | The fallback message comes from `CHART_ERROR_COPY.generic` (static i18n SSoT) by default. Callers may pass `fallbackMessage` prop, but the only Wave 2 caller (Plan 15-03) does NOT pass this prop — the default applies. Acceptance criteria verifies `import { CHART_ERROR_COPY }` is present. |
| T-15-02-02 | Information Disclosure (XSS via error message) | fallback `<p>` text rendering | high | mitigate | (a) `error.message` is NEVER rendered to user — only `CHART_ERROR_COPY.generic` (static) or trusted `fallbackMessage` is rendered. (b) No `dangerouslySetInnerHTML` in this file (AC: grep returns 0). (c) React auto-escapes JSX text content; an attacker who could craft an `error.message` containing `<script>...</script>` cannot surface it through this boundary's UI. AC #11 (zero `dangerouslySetInnerHTML`) is the grep-verifiable mitigation gate. **Block-on-high satisfied: high threat fully mitigated.** |
