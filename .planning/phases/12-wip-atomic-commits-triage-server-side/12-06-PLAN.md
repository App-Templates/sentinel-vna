```tsx
import { DollarSign, MousePointer, Eye, Percent } from 'lucide-react'
import { MetricCard } from '../ui/MetricCard'
import { CardSkeleton } from '../ui/Skeleton'
import { useRevenueStatus24h } from '../../api/client'

interface ExtendedRevenueData {
  rpm?: number
  ctr?: number
  viewability?: number
  gross_margin_pct?: number
}

export function KpiCards() {
  const { data, isLoading } = useRevenueStatus24h()

  if (isLoading) {
    return (
      <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
        {Array.from({ length: 4 }).map((_, i) => (
          <CardSkeleton key={i} />
        ))}
      </div>
    )
  }

  const ext = data as (typeof data & ExtendedRevenueData) | undefined

  const rpm = ext?.rpm ?? 0
  const ctr = ext?.ctr ?? 0
  const viewability = ext?.viewability ?? 0
  const grossMargin = ext?.gross_margin_pct ?? 0

  return (
    <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">
      <MetricCard
        label="RPM"
        value={rpm}
        format="currency"
        icon={<DollarSign size={20} />}
      />
      <MetricCard
        label="CTR"
        value={ctr}
        format="percent"
        icon={<MousePointer size={20} />}
      />
      <MetricCard
        label="Viewability"
        value={viewability}
        format="percent"
        icon={<Eye size={20} />}
      />
      <MetricCard
        label="Gross Margin"
        value={grossMargin}
        format="percent"
        icon={<Percent size={20} />}
      />
    </div>
  )
}
```
