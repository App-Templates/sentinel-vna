    import { useEffect, useRef, useState } from 'react'
    import { cn } from '../../lib/utils'

    interface DatePickerProps {
      value: string           // YYYY-MM-DD ISO date string
      availableDates: string[] // ISO dates, newest first; used for min/max bounds
      onChange: (date: string) => void
      className?: string
    }

    export function DatePicker({ value, availableDates, onChange, className }: DatePickerProps) {
      const [localValue, setLocalValue] = useState(value)
      const commitRef = useRef(value)

      useEffect(() => {
        setLocalValue(value)
        commitRef.current = value
      }, [value])

      const commit = () => {
        if (localValue !== commitRef.current && localValue) {
          commitRef.current = localValue
          onChange(localValue)
        }
      }

      const min = availableDates.length > 0 ? availableDates[availableDates.length - 1] : undefined
      const max = availableDates.length > 0 ? availableDates[0] : undefined

      return (
        <input
          type="date"
          value={localValue}
          min={min}
          max={max}
          onChange={(e) => setLocalValue(e.target.value)}
          onBlur={commit}
          onKeyDown={(e) => { if (e.key === 'Enter') { commit(); (e.currentTarget as HTMLInputElement).blur() } }}
          className={cn(
            'cursor-pointer rounded-md border border-gray-200 bg-gray-50 px-3 py-1.5 text-sm font-medium text-gray-700 shadow-sm transition-colors hover:bg-gray-100 focus:border-blue-500 focus:outline-none focus:ring-1 focus:ring-blue-500',
            className
          )}
        />
      )
    }
