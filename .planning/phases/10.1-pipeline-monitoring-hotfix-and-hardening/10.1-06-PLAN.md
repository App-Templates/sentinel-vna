Rewrite `send_meta_alert` retry loop per D-10.1-06 to fix **MAJOR-02**
(auditor 2): current code has dead branches because `urlopen()` raises
`HTTPError` on any non-2xx response; the `resp.status >= 500` branch and the
`else: return False` branch for non-200 are **unreachable in practice**.
Worse, HTTP 429 rate-limit responses — which `urlopen` wraps as
`HTTPError(code=429)` — fall through to `except (URLError, OSError)` where
they're retried without honoring `Retry-After`, so the backoff is wrong.

The correct pattern distinguishes:
- `HTTPError(429)` → read `Retry-After` header, sleep, retry
- `HTTPError(5xx)` → exponential backoff retry
- `HTTPError(4xx non-429)` → non-retriable, log + return False
- `URLError` (DNS, connection refused, read timeout) → retry with exponential backoff

```python
import urllib.error
# ... (existing imports)

def send_meta_alert(metrics: dict, _webhook_override=None) -> bool:
    webhook_url = _webhook_override or _resolve_webhook_url()
    if not webhook_url:
        logger.warning(...)
        return False

    blocks = _build_meta_alert_blocks(metrics)
    payload_json = json.dumps({"blocks": blocks}).encode("utf-8")
    max_retries = 3

    for attempt in range(max_retries):
        try:
            req = Request(webhook_url, data=payload_json, ..., method="POST")
            with urlopen(req, timeout=10) as resp:
                if resp.status == 200:
                    logger.info("send_meta_alert: Slack POST succeeded ...")
                    return True
                # Defensive path: non-urlopen-raising non-2xx (rare)
                logger.warning("send_meta_alert: Slack returned %d (non-raising)", resp.status)
                return False
        except urllib.error.HTTPError as e:
            if e.code == 429:
                retry_after = e.headers.get("Retry-After", "")
                try:
                    backoff = int(retry_after) if retry_after.isdigit() else (2 ** attempt)
                except Exception:
                    backoff = 2 ** attempt
                if attempt < max_retries - 1:
                    logger.warning(
                        "send_meta_alert: Slack 429 rate-limit (attempt %d/%d), sleeping %ds",
                        attempt + 1, max_retries, backoff,
                    )
                    time.sleep(backoff)
                    continue
                else:
                    logger.error("send_meta_alert: 429 after %d retries", max_retries)
                    return False
            elif e.code >= 500:
                if attempt < max_retries - 1:
                    delay = 2 ** attempt
                    logger.warning(
                        "send_meta_alert: Slack %d (attempt %d/%d), retrying in %ds",
                        e.code, attempt + 1, max_retries, delay,
                    )
                    time.sleep(delay)
                    continue
                else:
                    logger.error("send_meta_alert: 5xx after %d retries", max_retries)
                    return False
            else:
                logger.error(
                    "send_meta_alert: Slack %d %s non-retriable", e.code, e.reason,
                )
                return False
        except urllib.error.URLError as e:
            if attempt < max_retries - 1:
                delay = 2 ** attempt
                logger.warning(
                    "send_meta_alert: network error (attempt %d/%d), retrying in %ds: %r",
                    attempt + 1, max_retries, delay, e,
                )
                time.sleep(delay)
                continue
            else:
                logger.error("send_meta_alert: URLError after %d retries: %r", max_retries, e)
                return False
    return False
```
