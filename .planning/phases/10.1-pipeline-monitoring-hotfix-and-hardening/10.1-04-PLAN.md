        def test_temporal_query_misses_t_separator_rows_pre_migration(self):
            """Simulates pre-migration state: rows with T-separator would be
            compared against datetime('now','-1 day') output (space-separator).
            String comparison semantics: 'T' (0x54) > space (0x20), so if the
            cutoff is '2026-04-23 06:00:00' and a row is '2026-04-23T10:00:00',
            `'2026-04-23T10:00:00' >= '2026-04-23 06:00:00'` is TRUE (the 'T'
            position sorts above ' ', so the T-row SEEMS newer than it is).
            Actually — this means T-rows are OVER-counted as recent, not under-counted.
            For rows exactly 1 day old spanning the boundary, T-rows are kept when
            they should have been filtered out (and vice versa depending on char position).

            This test asserts the SAME string `created_at >= datetime('now','-1 day')`
            comparison returns a DIFFERENT count on T-rows vs canonical rows
            when the row's date portion equals the cutoff's date portion.
            """
            from backend.db import operations as ops
            # Seed a row whose 24h window status is AMBIGUOUS under T-vs-space compare:
            # create a T-separator row at 'now - 23h' (clearly within 24h), then a
            # canonical row at 'now - 23h' — both should count, but T-row behavior
            # under string compare against space-sep cutoff is the bug.
            from datetime import datetime, timedelta
            ts_naive = datetime.utcnow() - timedelta(hours=23)
            canonical = ts_naive.isoformat(sep=' ', timespec='microseconds')
            t_form = ts_naive.isoformat(sep='T', timespec='microseconds')
            with _tmpdb_with_schema() as db_path:
                self._seed_alert_raw(db_path, "t-form-1", t_form)
                self._seed_alert_raw(db_path, "canon-1", canonical)
                with patch("backend.db.database.DB_PATH", db_path):
                    result = ops.get_alerts_pipeline_health()
                # Query the raw SQLite to confirm the count split
                conn = sqlite3.connect(db_path)
                t_in_window = conn.execute(
                    "SELECT COUNT(*) FROM alerts WHERE created_at LIKE '%T%' AND created_at >= datetime('now','-1 day')"
